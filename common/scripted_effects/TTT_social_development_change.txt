#教育发展度上升
TTT_research_and_education_upgrade = {
	if = {
		limit = { has_idea = Golden_Age_Research }
		add_stability = 0.1
		add_political_power = 200
	}
	else_if = {
		limit = { has_idea = Higher_education }
		swap_ideas = {
			remove_idea = Higher_education
			add_idea = Golden_Age_Research
		}
	}
	else_if = {
		limit = { has_idea = Compulsory_education }
		swap_ideas = {
			remove_idea = Compulsory_education
			add_idea = Higher_education
		}
	}
	else_if = {
		limit = { has_idea = Vocational_education }
		swap_ideas = {
			remove_idea = Vocational_education
			add_idea = Compulsory_education
		}
	}
	else_if = {
		limit = { has_idea = Lack_education }
		swap_ideas = {
			remove_idea = Lack_education
			add_idea = Vocational_education
		}
	}
	else_if = {
		limit = { has_idea = ideological_imprint }
		swap_ideas = {
			remove_idea = ideological_imprint
			add_idea = Lack_education
		}
	}
	else_if = {
		limit = { has_idea = Illiteracy_everywhere }
		swap_ideas = {
			remove_idea = Illiteracy_everywhere
			add_idea = ideological_imprint
		}
	}
}
