}

NAT_tribal_council_is_open = {
	original_tag = NAT
	has_country_flag = NAT_tribal_council_open
	NOT = { has_global_flag = NAT_tribal_council_abandoned }
}

}

	has_global_flag = NAT_final_battle_started

}
